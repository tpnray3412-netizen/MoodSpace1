    <script>
        const skyBg = document.getElementById('skyBg');
        const rainContainer = document.getElementById('rainContainer');
        const cardsContainer = document.getElementById('cardsContainer');
        const floraContainer = document.getElementById('floraContainer');
        const scrollHint = document.getElementById('scrollHint');
        const diaryModal = document.getElementById('diaryModal');
        const diaryTitle = document.getElementById('diaryTitle');
        const diarySubtitle = document.getElementById('diarySubtitle');
        const diaryInput = document.getElementById('diaryInput');

        // สร้างเม็ดฝน
        for (let i = 0; i < 60; i++) {
            let drop = document.createElement('div');
            drop.className = 'raindrop';
            drop.style.left = Math.random() * 100 + 'vw';
            drop.style.top = (Math.random() * -50) + 'px';
            drop.style.animationDuration = (0.5 + Math.random() * 0.5) + 's';
            drop.style.animationDelay = (Math.random() * 2) + 's';
            rainContainer.appendChild(drop);
        }

        // จัดการ Scroll Storyตามระยะความสูงหน้าจอ
        window.addEventListener('scroll', () => {
            let scrollY = window.scrollY;
            let maxScroll = document.body.scrollHeight - window.innerHeight;
            let scrollPercent = scrollY / maxScroll;

            if (scrollPercent < 0.35) {
                // ฉากที่ 1: หน้าแรก (ฟ้าใส หญ้าเขียว)
                skyBg.style.background = 'linear-gradient(to bottom, #4facfe 0%, #00f2fe 100%)';
                rainContainer.style.opacity = '0';
                floraContainer.style.opacity = '0';
                floraContainer.style.transform = 'scale(0.8)';
                cardsContainer.style.opacity = '0';
                cardsContainer.style.pointerEvents = 'none';
                scrollHint.innerText = '📜 เลื่อนลงเพื่อสัมผัสสายฝน...';
            } 
            else if (scrollPercent >= 0.35 && scrollPercent < 0.7) {
                // ฉากที่ 2: ฝนตก
                skyBg.style.background = 'linear-gradient(to bottom, #3a6073 0%, #16222a 100%)';
                rainContainer.style.opacity = '1';
                floraContainer.style.opacity = '0';
                cardsContainer.style.opacity = '0';
                cardsContainer.style.pointerEvents = 'none';
                scrollHint.innerText = '📜 เลื่อนลงเพื่อดูธรรมชาติเติบโต...';
            } 
            else {
                // ฉากที่ 3: ฝนหยุด ดอกไม้โต + การ์ดลอย
                skyBg.style.background = 'linear-gradient(to bottom, #89f7fe 0%, #66a6ff 100%)';
                rainContainer.style.opacity = '0';
                floraContainer.style.opacity = '1';
                floraContainer.style.transform = 'scale(1)';
                cardsContainer.style.opacity = '1';
                cardsContainer.style.pointerEvents = 'auto';
                scrollHint.innerText = '✨ คลิกเลือกการ์ดอารมณ์ด้านบนเพื่อเขียน Diary ได้เลย!';
            }
        });

        // เปิด-ปิด Modal Diary
        function openDiary(emotionName, subtitleText) {
            diaryTitle.innerText = `บันทึกอารมณ์: ${emotionName}`;
            diarySubtitle.innerText = subtitleText;
            diaryModal.classList.add('active');
        }

        function closeDiary() {
            diaryModal.classList.remove('active');
            diaryInput.value = '';
        }

        function saveDiary() {
            if(diaryInput.value.trim() === "") {
                alert("กรุณากรอกข้อความความรู้สึกก่อนบันทึกนะคะ");
                return;
            }
            alert("บันทึกความรู้สึกของคุณเรียบร้อยแล้ว ขอให้วันนี้เป็นวันที่ดีนะคับ 💚");
            closeDiary();
        }
    </script>
